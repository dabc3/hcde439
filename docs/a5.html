Calculations:

For the photoresistor, I took measurements of the resistance in both the light (Rlow) and the dark (Rhigh) using my multimeter, and used the equation R2 = sqrt(Rlow x Rhigh). I then rounded to the nearest resistor which would be 10K ohms. This was the same process I used for the thermistor, but instead of light and dark conditions, I measured the resistance when I was touching it and not.

Code Snippet:

          #include < IRremote.hpp> //import IRremote library

          const int IRreceiverPin = 2; // pin that holds the IR receiver
          const int LEDsPin = 10; // pin that holds the leds
          int brightness = 0; // how bright the LED is
          int fadeAmount = 0; // how many points to fade the LED by

          // setup components
          void setup() {
            IrReceiver.begin(IRreceiverPin, ENABLE_LED_FEEDBACK); // allow the receiver to give led feedback from remote
            pinMode(LEDsPin, OUTPUT); // make the ledspin int an output
            }

          void loop() {
            // take output from IR reciever
            if (IrReceiver.decode()){
              // if button 1 is pressed
              if (IrReceiver.decodedIRData.decodedRawData == 0xF30CFF00){ 
                // change the fade amount to 5
                fadeAmount = 5;
              }
              // if button 2 is pressed 
              if (IrReceiver.decodedIRData.decodedRawData == 0xE718FF00){
                // make brightness 0 (turn leds off)
                brightness = 0;
                // make fade amount 0 (do not fade leds)
                fadeAmount = 0;
              }
              // resume the decoder
              IrReceiver.resume();
            }
            // Only run the fade logic if fadeAmount is not 0
            if (fadeAmount != 0) {
              // add fade amount to brightness value
              brightness += fadeAmount;

              // Reverse direction and constrain brightness
              if (brightness <= 0 || brightness >= 255) {
                // make fade negative so it fades the opposite direction
                fadeAmount = -fadeAmount;
              }
              
              // keep brightness within 0-255 range
              brightness = constrain(brightness, 0, 255);
              // set the LEDsPin output to the brightness value
              analogWrite(LEDsPin, brightness);
              
              delay(30); // Control the speed of the fade
            } else {
              // If not fading, ensure it stays off
              analogWrite(LEDsPin, brightness);
            }
          }
